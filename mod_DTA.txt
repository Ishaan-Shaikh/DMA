def entropy(labels):
    counts, total = Counter(labels), len(labels)
    return -sum((c / total) * log2(c / total) for c in counts.values())


def split_stats(rows, feature, target):
    """Return (information gain, split info) for splitting on `feature`."""
    n = len(rows)
    weighted = 0.0
    split_info = 0.0
    for value in {r[feature] for r in rows}:
        subset = [r for r in rows if r[feature] == value]
        w = len(subset) / n
        weighted += w * entropy([r[target] for r in subset])
        split_info -= w * log2(w)
    gain = entropy([r[target] for r in rows]) - weighted
    return gain, split_info


def score(rows, feature, target, criterion):
    gain, split_info = split_stats(rows, feature, target)
    if criterion == "gain":                 # original ID3
        return gain
    if split_info == 0:                     # feature has a single value: useless
        return 0.0
    return gain / split_info                # MODIFICATION: gain ratio


def build(rows, features, target, criterion, min_gain=0.0, verbose=False, depth=0):
    labels = [r[target] for r in rows]
    majority = Counter(labels).most_common(1)[0][0]
    if len(set(labels)) == 1 or not features:
        return majority

    scores = {f: score(rows, f, target, criterion) for f in features}
    best = max(scores, key=scores.get)
    if verbose:
        pad = "    " * depth
        print(f"{pad}n={len(rows)}  " + "  ".join(f"{f}={s:.3f}" for f, s in scores.items())
              + f"  -> split on {best}")
    if scores[best] <= min_gain:
        return majority

    rest = [f for f in features if f != best]
    node = {best: {}}
    for v in sorted({r[best] for r in rows}):
        sub = [r for r in rows if r[best] == v]
        node[best][v] = build(sub, rest, target, criterion, min_gain, verbose, depth + 1)
    node[best]["_default"] = majority
    return node


def predict(tree, ex):
    while isinstance(tree, dict):
        f = next(iter(tree))
        tree = tree[f].get(ex.get(f), tree[f]["_default"])
    return tree


def show(tree, indent=""):
    if not isinstance(tree, dict):
        print(f"{indent}-> {tree}"); return
    f = next(iter(tree))
    for v, sub in tree[f].items():
        if v != "_default":
            print(f"{indent}{f} = {v}:"); show(sub, indent + "    ")


def loocv(rows, features, criterion):
    hits = 0
    for i, test in enumerate(rows):
        train = rows[:i] + rows[i + 1:]
        hits += predict(build(train, features, TARGET, criterion), test) == test[TARGET]
    return hits / len(rows)


def print_table(rows):
    """Print the whole dataset as an aligned table."""
    cols = list(rows[0])
    widths = {c: max(len(c), *(len(r[c]) for r in rows)) for c in cols}
    line = "+-" + "-+-".join("-" * widths[c] for c in cols) + "-+"
    print(line)
    print("| " + " | ".join(f"{c:<{widths[c]}}" for c in cols) + " |")
    print(line)
    for r in rows:
        print("| " + " | ".join(f"{r[c]:<{widths[c]}}" for c in cols) + " |")
    print(line)
