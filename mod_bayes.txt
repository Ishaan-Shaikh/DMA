"""Improved Naive Bayesian Classifier:
   1. Laplace (add-k) smoothing  -> no zero-probability problem
   2. Log-probabilities          -> no numerical underflow
   3. Normalised posteriors      -> real P(class | X) values that sum to 1
   4. Reusable class with fit / predict_proba / predict
"""
import math
from collections import Counter, defaultdict

HEADER = ["RID", "Age", "Income", "Student", "Credit_rating", "buys_computer"]
DATA = [
    (1,  "<=30",   "high",   "no",  "fair",  "no"),
    (2,  "<=30",   "high",   "no",  "excel", "no"),
    (3,  "31..40", "high",   "no",  "fair",  "yes"),
    (4,  ">40",    "medium", "no",  "fair",  "yes"),
    (5,  ">40",    "low",    "yes", "fair",  "yes"),
    (6,  ">40",    "low",    "yes", "excel", "no"),
    (7,  "31..40", "low",    "yes", "excel", "yes"),
    (8,  "<=30",   "medium", "no",  "fair",  "no"),
    (9,  "<=30",   "low",    "yes", "fair",  "yes"),
    (10, ">40",    "medium", "yes", "fair",  "yes"),
    (11, "<=30",   "medium", "yes", "excel", "yes"),
    (12, "31..40", "medium", "no",  "excel", "yes"),
    (13, "31..40", "high",   "yes", "fair",  "yes"),
    (14, ">40",    "medium", "no",  "excel", "no"),
]
ATTRS = HEADER[1:-1]
TEST_X = {"Age": "<=30", "Income": "medium", "Student": "yes", "Credit_rating": "fair"}
# Age = 31..40 never occurs with class 'no' -> plain Bayes gives probability 0
TEST_X2 = {"Age": "31..40", "Income": "medium", "Student": "no", "Credit_rating": "excel"}


def print_table(title, header, rows):
    print(title)
    widths = [max(len(str(x)) for x in col) for col in zip(header, *rows)]
    line = "+" + "+".join("-" * (w + 2) for w in widths) + "+"
    fmt = lambda r: "| " + " | ".join(str(v).ljust(w) for v, w in zip(r, widths)) + " |"
    print(line); print(fmt(header)); print(line)
    for r in rows:
        print(fmt(r))
    print(line)


class ImprovedNaiveBayes:
    def __init__(self, alpha=1.0):
        self.alpha = alpha  # Laplace smoothing; alpha=0 -> plain Bayes

    def fit(self, rows, attrs):
        self.attrs = attrs
        labels = [r[-1] for r in rows]
        self.n = len(rows)
        self.class_count = Counter(labels)
        self.classes = sorted(self.class_count)
        self.counts = defaultdict(Counter)       # (class, attr) -> value counts
        self.values = {a: set() for a in attrs}  # distinct values per attribute
        for r in rows:
            for i, a in enumerate(attrs, start=1):
                self.counts[(r[-1], a)][r[i]] += 1
                self.values[a].add(r[i])
        return self

    def cond_prob(self, a, v, c):
        k = len(self.values[a])
        return (self.counts[(c, a)][v] + self.alpha) / (self.class_count[c] + self.alpha * k)

    def predict_proba(self, x, verbose=False):
        log_scores = {}
        for c in self.classes:
            prior = self.class_count[c] / self.n
            if verbose:
                print(f"\nClass '{c}':  P({c}) = {self.class_count[c]}/{self.n} = {prior:.3f}")
            logp = math.log(prior)
            for a in self.attrs:
                p = self.cond_prob(a, x[a], c)
                if verbose:
                    cnt = self.counts[(c, a)][x[a]]
                    k = len(self.values[a])
                    print(f"   P({a} = {x[a]!r} | {c}) = ({cnt}+{self.alpha:g})/"
                          f"({self.class_count[c]}+{self.alpha:g}*{k}) = {p:.3f}")
                logp += math.log(p) if p > 0 else float("-inf")
            log_scores[c] = logp
        m = max(log_scores.values())
        exp = {c: math.exp(s - m) for c, s in log_scores.items()}
        z = sum(exp.values())
        return {c: v / z for c, v in exp.items()}, log_scores

    def predict(self, x, verbose=False):
        probs, _ = self.predict_proba(x, verbose)
        return max(probs, key=probs.get), probs


def show_result(model, x, title):
    print(f"\n=== {title} ===")
    print_table("Test sample X", ATTRS, [[x[a] for a in ATTRS]])
    label, probs = model.predict(x, verbose=True)
    print("\nPosterior probabilities P(class | X):")
    for c, p in probs.items():
        print(f"   P({c} | X) = {p:.4f}")
    print(f"Prediction: buys_computer = '{label}'")


if __name__ == "__main__":
    print_table("TRAINING DATASET (buys_computer)", HEADER, DATA)
    print_table("\nTEST SAMPLES", ["Sample"] + ATTRS,
                [["X1"] + [TEST_X[a] for a in ATTRS], ["X2"] + [TEST_X2[a] for a in ATTRS]])

    model = ImprovedNaiveBayes(alpha=1.0).fit(DATA, ATTRS)
    show_result(model, TEST_X, "X1 with Laplace smoothing")

    # Why smoothing matters: X2 has a zero count for class 'no'
    plain = ImprovedNaiveBayes(alpha=0).fit(DATA, ATTRS)
    print("\n=== X2 WITHOUT smoothing (plain Bayes) ===")
    print("P(Age='31..40' | no) = 0/5 = 0  ->  whole product for 'no' becomes 0")
    print("Result:", plain.predict(TEST_X2)[1])
    show_result(model, TEST_X2, "X2 WITH Laplace smoothing")
