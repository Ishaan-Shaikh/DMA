"""Naive Bayesian Classifier (basic) - buys_computer example."""

HEADER = ["RID", "Age", "Income", "Student", "Credit_rating", "buys_computer"]
DATA = [
    (1,  "<=30",   "high",   "no",  "fair",  "no"),
    (2,  "<=30",   "high",   "no",  "excel", "no"),
    (3,  "31..40", "high",   "no",  "fair",  "yes"),
    (4,  ">40",    "medium", "no",  "fair",  "yes"),
    (5,  ">40",    "low",    "yes", "fair",  "yes"),
    (6,  ">40",    "low",    "yes", "excel", "no"),
    (7,  "31..40", "low",    "yes", "excel", "yes"),
    (8,  "<=30",   "medium", "no",  "fair",  "no"),
    (9,  "<=30",   "low",    "yes", "fair",  "yes"),
    (10, ">40",    "medium", "yes", "fair",  "yes"),
    (11, "<=30",   "medium", "yes", "excel", "yes"),
    (12, "31..40", "medium", "no",  "excel", "yes"),
    (13, "31..40", "high",   "yes", "fair",  "yes"),
    (14, ">40",    "medium", "no",  "excel", "no"),
]
ATTRS = HEADER[1:-1]
TEST_X = {"Age": "<=30", "Income": "medium", "Student": "yes", "Credit_rating": "fair"}


def print_table(title, header, rows):
    print(title)
    widths = [max(len(str(x)) for x in col) for col in zip(header, *rows)]
    line = "+" + "+".join("-" * (w + 2) for w in widths) + "+"
    fmt = lambda r: "| " + " | ".join(str(v).ljust(w) for v, w in zip(r, widths)) + " |"
    print(line); print(fmt(header)); print(line)
    for r in rows:
        print(fmt(r))
    print(line)


def naive_bayes(data, x):
    classes = sorted({r[-1] for r in data})
    total = len(data)
    scores = {}
    for c in classes:
        rows_c = [r for r in data if r[-1] == c]
        prior = len(rows_c) / total
        print(f"\nClass '{c}':  P({c}) = {len(rows_c)}/{total} = {prior:.3f}")
        likelihood = 1.0
        for i, a in enumerate(ATTRS, start=1):
            count = sum(1 for r in rows_c if r[i] == x[a])
            p = count / len(rows_c)
            print(f"   P({a} = {x[a]!r} | {c}) = {count}/{len(rows_c)} = {p:.3f}")
            likelihood *= p
        scores[c] = likelihood * prior
        print(f"   P(X | {c}) = {likelihood:.3f}")
        print(f"   P(X | {c}) * P({c}) = {scores[c]:.3f}")
    return scores


if __name__ == "__main__":
    print_table("TRAINING DATASET (buys_computer)", HEADER, DATA)
    print_table("\nTEST SAMPLE X", ATTRS, [[TEST_X[a] for a in ATTRS]])

    print("\n--- Naive Bayesian Classification ---")
    scores = naive_bayes(DATA, TEST_X)
    prediction = max(scores, key=scores.get)
    print(f"\nPrediction: buys_computer = '{prediction}' for sample X")
